A bulging curve between two joints — height grows as they compress, with optional spring overshoot.

Inputs
Joint A & B positions, max height, profile exponent, apex slide, spring blend / stiffness / damping
Output
Curve points along the span — portable to any path, CV list, or joint chain

As distance shrinks, target bulge rises. A superellipse profile shapes the peak; sliding the apex shifts bulk left or right. Blend in a damped spring so the bulge lags and settles instead of snapping.

Pseudocode 
distance = length(jointB - jointA)
compression = (maxDist - distance) / (maxDist - minDist)
h_target = maxHeight * clamp(compression, 0, 1)
h = lerp(h_target, spring_integrate(h_target), springBlend)

Full formula breakdown 
1. Harmonic Spring-Mass Oscillator Simulation:
   h_kinematic = max_h * (1 - Distance / Max_Distance)
   a_spring = -Stiffness * (h_current - h_target) - (2 * Damping * sqrt(Stiffness)) * v_current
   v_next = v_current + a_spring * dt
   h_spring = h_current + v_next * dt
   h_final = lerp(h_kinematic, h_spring, Blend)

2. Piecewise Linear Slide & Superellipse Profile:
   if u < apexSlide:
       u_warped = (u - apexSlide) / (1.0 + apexSlide)
   else:
       u_warped = (u - apexSlide) / (1.0 - apexSlide)

   profile = (1 - |u_warped|^n)^(1/n)
   y(u) = base_y - h_final * profile

import maya.cmds as cmds

def build_muscle_rig():
    name = "MuscleCurveRig"
    if cmds.objExists(name): cmds.delete(name)
    
    root = cmds.group(em=True, name=name)
    
    # 1. Create Baseline Joints
    jA = cmds.joint(name=f"{name}_JointA", p=(-210, 0, 0))
    jB = cmds.joint(name=f"{name}_JointB", p=(210, 0, 0))
    
    # 2. Muscle Curve (Rest State Flat on X-Axis)
    cvs = [(-210 + (420 * (i/11.0)), 0, 0) for i in range(12)]
    muscle_crv = cmds.curve(name=f"{name}_MuscleCurve", d=3, p=cvs)
    cmds.parent(muscle_crv, root)
    
    # 3. Parameter Controls
    settings = cmds.spaceLocator(name=f"{name}_Settings")[0]
    cmds.parent(settings, root)
    cmds.setAttr(f"{settings}.visibility", 0)
    
    attrs = [
        ("maxHeight", 95.0, -80.0, 140.0),
        ("exponent", 2.0, 0.4, 4.0),
        ("apexSlide", 0.0, -0.85, 0.85)
    ]
    for attr, dv, minv, maxv in attrs:
        cmds.addAttr(settings, ln=attr, at="float", dv=dv, min=minv, max=maxv, k=True)
    
    # 4. Kinematic Distance
    dist_node = cmds.createNode('distanceBetween', name=f"{name}_Dist")
    cmds.connectAttr(f"{jA}.worldPosition[0]", f"{dist_node}.point1")
    cmds.connectAttr(f"{jB}.worldPosition[0]", f"{dist_node}.point2")
    
    # 5. Generate Unrolled Piecewise Expression (Highly Optimized)
    expr_lines = [
        f"// --- Kinematic Height Base ---",
        f"float $dist = {dist_node}.distance;",
        f"float $contraction = max(0.0, 420.0 - $dist);",
        f"float $hFinal = {settings}.maxHeight * ($contraction / 340.0);",
        f"",
        f"// --- Piecewise Profile Variables ---",
        f"float $slide = {settings}.apexSlide;",
        f"float $n = {settings}.exponent;"
    ]
    
    # Python procedurally unrolls the piecewise math for all 12 Control Vertices
    for i in range(12):
        u = -1.0 + (i / 11.0) * 2.0
        expr_lines.extend([
            f"",
            f"// CV {i} (Base U = {u:.3f})",
            f"float $u{i} = {u:.3f};",
            f"float $w{i} = ($u{i} < $slide) ? ($u{i} - $slide)/(1.0 + $slide) : ($u{i} - $slide)/(1.0 - $slide);",
            f"float $p{i} = pow(max(0.0, 1.0 - pow(abs($w{i}), $n)), 1.0 / $n);",
            f"{muscle_crv}.controlPoints[{i}].yValue = $hFinal * $p{i};"
        ])
        
    # Inject exact match math directly into the dependency graph
    cmds.expression(s="\\n".join(expr_lines), name=f"{name}_DeformExpr")
    
    cmds.select(settings)
    print("Piecewise linear muscle rig generated.")

build_muscle_rig()
